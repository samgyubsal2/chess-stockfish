// ========================================== // STOCKFISH CHESS ANALYSIS // ========================================== const analyzeButton = document.getElementById("analyze"); const analysisPanel = document.getElementById("analysis-panel"); const analysisSummary = document.getElementById("analysis-summary"); const analysisResults = document.getElementById("analysis-results"); const analysisProgress = document.getElementById("analysis-progress"); const ENGINE_PATH = "stockfish-19-lite-single.js"; const ANALYSIS_DEPTH = 12; let stockfish = null; let engineReady = false; let engineInitializing = false; let currentSearch = null; // Initialize Stockfish in a Web Worker. function initializeStockfish() { if (stockfish || engineInitializing) return; engineInitializing = true; analysisProgress.textContent = "Loading Stockfish..."; try { stockfish = new Worker(ENGINE_PATH); } catch (error) { engineInitializing = false; analysisProgress.textContent = "Could not start Stockfish. Check the engine files and local server."; console.error(error); return; } stockfish.onmessage = function(event) { const line = String(event.data).trim(); // Engine initialization. if (line === "uciok") { stockfish.postMessage("isready"); return; } if (line === "readyok") { engineReady = true; engineInitializing = false; analysisProgress.textContent = "Stockfish is ready."; analyzeButton.disabled = !game.game_over(); return; } // Save the latest evaluation reported during the search. if (currentSearch && line.startsWith("info ")) { const depthMatch = line.match(/\bdepth\s+(\d+)/); const scoreMatch = line.match( /\bscore\s+(cp|mate)\s+(-?\d+)\b/ ); // Ignore incomplete bounds and evaluations from shallower depths. if ( depthMatch && scoreMatch && !/\b(?:lowerbound|upperbound)\b/.test(line) ) { const depth = Number(depthMatch[1]); if ( depth >= currentSearch.depth && (!currentSearch.evaluation || depth >= currentSearch.evaluation.depth) ) { currentSearch.evaluation = { depth, type: scoreMatch[1], value: Number(scoreMatch[2]) }; } } } // A completed search returns its best move. if (currentSearch && line.startsWith("bestmove ")) { const search = currentSearch; currentSearch = null; clearTimeout(search.timeout); search.resolve({ bestMove: line.split(/\s+/)[1], evaluation: search.evaluation }); } }; stockfish.onerror = function(error) { engineReady = false; engineInitializing = false; analysisProgress.textContent = "Stockfish failed to load. Check the JavaScript and WASM files."; console.error("Stockfish worker error:", error); }; stockfish.postMessage("uci"); } // Convert a Stockfish score into centipawns. // // Stockfish reports scores from the perspective of the // player whose turn it is in the analyzed position. function scoreToCentipawns(evaluation) { if (!evaluation) return null; if (evaluation.type === "cp") { return evaluation.value; } // Represent mate scores with very large signed values. const mate = evaluation.value; if (mate > 0) return 100000 - mate * 100; if (mate < 0) return -100000 - mate * 100; return 0; } // Analyze one FEN position and return Stockfish's best move // and its evaluation. function analyzePosition(fen, depth = ANALYSIS_DEPTH) { return new Promise((resolve, reject) => { if (!engineReady || !stockfish) { reject(new Error("Stockfish is not ready.")); return; } if (currentSearch) { reject(new Error("Another engine search is already running.")); return; } const search = { resolve, reject, depth, evaluation: null, timeout: null }; currentSearch = search; search.timeout = setTimeout(() => { if (currentSearch !== search) return; currentSearch = null; reject(new Error("Stockfish search timed out.")); }, 120000); stockfish.postMessage("position fen " + fen); stockfish.postMessage("go depth " + depth); }); } // Convert a chess.js move into UCI notation. function toUCI(move) { return move.from + move.to + (move.promotion || ""); } // Evaluate a position from the perspective of a particular player. function scoreForPlayer(fen, evaluation, playerColor) { const score = scoreToCentipawns(evaluation); if (score === null) return null; const sideToMove = fen.split(/\s+/)[1]; const whitePerspective = sideToMove === "w" ? score : -score; return playerColor === "w" ? whitePerspective : -whitePerspective; } // Assign approximate move labels based on the centipawn loss. function classifyStockfishMove(loss, move, isBestMove) { if (move.san.includes("#")) return "Checkmate"; if (loss === null) return "Unclassified"; if (loss <= 10) { return isBestMove ? "Best" : "Excellent"; } if (loss <= 30) return "Excellent"; if (loss <= 60) return "Good"; if (loss <= 100) return "Inaccuracy"; if (loss <= 200) return "Mistake"; return "Blunder"; } // Give a basic explanation for the result. function explainStockfishMove(record) { if (record.move.san.includes("#")) { return "This move delivers checkmate."; } if (record.isBestMove) { return "This matches Stockfish's recommended move at the selected depth."; } if (record.loss === null) { return "Stockfish did not return enough evaluation data."; } if (record.loss <= 30) { return "Very close to Stockfish's preferred evaluation."; } if (record.loss <= 100) { return "Stockfish evaluates another move as noticeably stronger."; } return "Stockfish found a substantially stronger alternative."; } // Reconstruct the game and analyze every move. async function analyzeGame() { if (!game.game_over()) { alert("Finish the game before analyzing it."); return; } if (!engineReady) { initializeStockfish(); analysisProgress.textContent = "Waiting for Stockfish to initialize..."; return; } analyzeButton.disabled = true; analysisPanel.style.display = "block"; analysisResults.replaceChildren(); analysisSummary.replaceChildren(); const replay = new Chess(); const history = game.history({ verbose: true }); const records = []; try { for (let i = 0; i < history.length; i++) { const move = history[i]; const beforeFen = replay.fen(); const mover = replay.turn(); analysisProgress.textContent = `Analyzing move ${i + 1} of ${history.length}...`; // Find Stockfish's preferred move before the actual move. const before = await analyzePosition(beforeFen); // Play the actual move to reconstruct the resulting position. const played = replay.move({ from: move.from, to: move.to, promotion: move.promotion }); if (!played) { throw new Error("Could not replay move " + (i + 1)); } const afterFen = replay.fen(); // Analyze the position after the actual move. const after = await analyzePosition(afterFen); const bestScore = scoreForPlayer( beforeFen, before.evaluation, mover ); // The side to move after the actual move is the opponent, // so convert the score to the original mover's perspective. const actualScore = scoreForPlayer( afterFen, after.evaluation, mover ); const loss = bestScore === null || actualScore === null ? null : Math.max(0, bestScore - actualScore); const bestMove = before.bestMove; const isBestMove = bestMove !== "(none)" && bestMove === toUCI(move); records.push({ number: Math.floor(i / 2) + 1, color: mover, move, bestMove, loss, isBestMove, classification: classifyStockfishMove( loss, move, isBestMove ) }); // Yield to the browser so progress can update. await new Promise(resolve => setTimeout(resolve, 0)); } displayStockfishAnalysis(records); analysisProgress.textContent = "Analysis complete. Stockfish depth: " + ANALYSIS_DEPTH + "."; } catch (error) { analysisProgress.textContent = "Analysis failed: " + error.message; console.error(error); } finally { analyzeButton.disabled = false; } } // Display all move-by-move results. function displayStockfishAnalysis(records) { analysisSummary.replaceChildren(); analysisResults.replaceChildren(); const counts = {}; for (const record of records) { counts[record.classification] = (counts[record.classification] || 0) + 1; } const summary = document.createElement("p"); summary.textContent = "Analyzed " + records.length + " moves. " + Object.entries(counts) .map(([label, count]) => label + ": " + count) .join(" | "); analysisSummary.appendChild(summary); for (const record of records) { const row = document.createElement("div"); row.style.cssText = [ "padding:10px", "margin:7px 0", "border-radius:6px", "background:#303846", "line-height:1.5" ].join(";"); const heading = document.createElement("strong"); heading.textContent = record.number + (record.color === "w" ? "." : "...") + " " + record.move.san; const label = document.createElement("span"); label.style.cssText = "float:right;color:#a5e6a5;font-weight:bold"; label.textContent = record.classification; const details = document.createElement("div"); details.style.cssText = "font-size:12px;color:#a9b0bd;clear:both"; details.textContent = "Best move: " + record.bestMove + " | Estimated loss: " + (record.loss === null ? "Unavailable" : Math.round(record.loss) + " cp"); const explanation = document.createElement("div"); explanation.style.cssText = "font-size:13px;margin-top:5px"; explanation.textContent = explainStockfishMove(record); row.append(heading, label, details, explanation); analysisResults.appendChild(row); } } // Initialize the engine when the page loads. initializeStockfish(); // Enable the button when the game ends. const originalRender = render; render = function() { originalRender(); analyzeButton.disabled = !game.game_over() || !engineReady; }; analyzeButton.addEventListener("click", analyzeGame);
